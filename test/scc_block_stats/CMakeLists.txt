cmake_minimum_required(VERSION 3.16)
project(vvenc_scc_dispatch_test LANGUAGES CXX)
# Focused integration test: compile the real buffer constructor, runtime CPU
# selector, dispatcher and SIMD implementations without rebuilding the encoder.
option(SCC_TEST_ENABLE_SIMD "Build the real x86 SIMD buffer dispatch" ON)
get_filename_component(VVENC_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(LIB "${VVENC_ROOT}/source/Lib")
set(VVENC_USE_UNSTABLE_API 0)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/vvenc")
configure_file("${VVENC_ROOT}/include/vvenc/vvenc.h.in" "${CMAKE_CURRENT_BINARY_DIR}/vvenc/vvenc.h" @ONLY)
add_executable(scc_dispatch_test scc_dispatch_test.cpp "${LIB}/CommonLib/Buffer.cpp")
target_compile_features(scc_dispatch_test PRIVATE cxx_std_14)
target_include_directories(scc_dispatch_test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${VVENC_ROOT}/include" "${LIB}" "${LIB}/CommonLib" "${LIB}/CommonLib/x86" "${LIB}/EncoderLib" "${LIB}/DecoderLib" "${LIB}/Utilities" "${VVENC_ROOT}/thirdparty")
if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang" OR NOT UNIX)
  message(FATAL_ERROR "This focused guard-page test requires Linux GCC/Clang")
endif()
target_compile_options(scc_dispatch_test PRIVATE -O2 -Wall -Wextra -Wno-unused-parameter -Wno-sign-compare -ffunction-sections -fdata-sections)
target_link_options(scc_dispatch_test PRIVATE -Wl,--gc-sections)
if(SCC_TEST_ENABLE_SIMD)
  target_compile_definitions(scc_dispatch_test PRIVATE TARGET_SIMD_X86=1)
  target_sources(scc_dispatch_test PRIVATE "${LIB}/CommonLib/x86/CommonDefX86.cpp" "${LIB}/CommonLib/x86/InitX86.cpp" "${LIB}/CommonLib/x86/sse41/Buffer_sse41.cpp" "${LIB}/CommonLib/x86/avx2/Buffer_avx2.cpp")
  set_source_files_properties("${LIB}/CommonLib/x86/CommonDefX86.cpp" PROPERTIES COMPILE_OPTIONS -mxsave)
  set_source_files_properties("${LIB}/CommonLib/x86/sse41/Buffer_sse41.cpp" PROPERTIES COMPILE_OPTIONS -msse4.1 COMPILE_DEFINITIONS USE_SSE41)
  set_source_files_properties("${LIB}/CommonLib/x86/avx2/Buffer_avx2.cpp" PROPERTIES COMPILE_OPTIONS -mavx2 COMPILE_DEFINITIONS USE_AVX2)
else()
  # Same backend-disable mechanism as VVENC_ENABLE_X86_SIMD=OFF: no
  # TARGET_SIMD_X86 or SIMD translation units. Disable compiler auto-vectorizing
  # this focused build too, so it cannot accidentally test the SIMD helper.
  target_compile_options(scc_dispatch_test PRIVATE -fno-tree-vectorize)
endif()
enable_testing()
if(SCC_TEST_ENABLE_SIMD)
  add_test(NAME scc_dispatch_auto COMMAND scc_dispatch_test AUTO)
  add_test(NAME scc_dispatch_runtime_scalar COMMAND scc_dispatch_test SCALAR)
else()
  add_test(NAME scc_dispatch_simd_disabled COMMAND scc_dispatch_test DISABLED)
endif()
