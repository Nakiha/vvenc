cmake_minimum_required(VERSION 3.16)
project(live_cpu_ledger_regression LANGUAGES CXX)
get_filename_component(VVENC_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(VVENC_USE_UNSTABLE_API 0)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/vvenc")
configure_file("${VVENC_ROOT}/include/vvenc/vvenc.h.in" "${CMAKE_CURRENT_BINARY_DIR}/vvenc/vvenc.h" @ONLY)
find_package(Threads REQUIRED)
add_executable(live_cpu_ledger_test ledger_test.cpp "${VVENC_ROOT}/source/Lib/Utilities/NoMallocThreadPool.cpp")
target_compile_features(live_cpu_ledger_test PRIVATE cxx_std_14)
target_compile_options(live_cpu_ledger_test PRIVATE -O2 -g -Wall -Wextra)
target_include_directories(live_cpu_ledger_test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${VVENC_ROOT}/include" "${VVENC_ROOT}/source/Lib")
target_compile_definitions(live_cpu_ledger_test PRIVATE HAVE_PTHREADS)
target_link_libraries(live_cpu_ledger_test PRIVATE Threads::Threads)
add_executable(live_cpu_snapshot_test ledger_snapshot_test.cpp)
target_compile_features(live_cpu_snapshot_test PRIVATE cxx_std_14)
target_compile_options(live_cpu_snapshot_test PRIVATE -O2 -g -Wall -Wextra)
target_include_directories(live_cpu_snapshot_test PRIVATE "${VVENC_ROOT}/source/Lib")
target_link_libraries(live_cpu_snapshot_test PRIVATE Threads::Threads)
enable_testing()
add_test(NAME cpu_ledger_enabled COMMAND live_cpu_ledger_test enabled)
add_test(NAME cpu_ledger_disabled COMMAND live_cpu_ledger_test disabled)
add_test(NAME cpu_ledger_concurrent_snapshot COMMAND live_cpu_snapshot_test)
set_tests_properties(cpu_ledger_enabled cpu_ledger_disabled cpu_ledger_concurrent_snapshot PROPERTIES TIMEOUT 20)
